(async () => {
 const base=process.argv[1] || 'http://127.0.0.1:3000';
 if(!['http://127.0.0.1:3000','https://sutra.purama.dev'].includes(base))throw Error('Unexpected probe target');
 const canonical='https://sutra.purama.dev';
 const forged={'x-forwarded-host':'untrusted.invalid','x-forwarded-proto':'https'};
 const cases=[
  {path:'/auth/callback',status:307,location:canonical+'/login?error=missing_code'},
  {path:'/auth/callback?next=https://untrusted.invalid',status:307,location:canonical+'/login?error=missing_code'},
  {path:'/auth/callback',headers:forged,status:307,location:canonical+'/login?error=missing_code'},
  {path:'/auth/callback?code=postdeploy-invalid-code',status:307,location:canonical+'/login?error=callback_failed'},
  {path:'/dashboard?postdeploy=1',status:307,location:canonical+'/login?next=%2Fdashboard%3Fpostdeploy%3D1'},
  {path:'/login',status:200},
  {path:'/api/social/connect',method:'POST',origin:canonical,status:401},
  {path:'/api/social/connect',method:'POST',origin:canonical,headers:forged,status:401},
  {path:'/api/social/connect',method:'POST',origin:'https://untrusted.invalid',status:403},
  {path:'/api/social/connect',method:'POST',status:401},
  {path:'/api/social/callback?error=postdeploy-invalid',status:302,location:canonical+'/settings/social?error=oauth_denied'},
  ...['/api/create','/api/production','/api/video/export'].map(path=>({path,method:'POST',origin:canonical,status:401})),
 ];
 const checks=[];
 for(const c of cases){
  const response=await fetch(base+c.path,{method:c.method||'GET',headers:{...c.headers,...(c.origin?{origin:c.origin}:{}),...(c.method?{'content-type':'application/json'}:{})},...(c.method?{body:'{}'}:{}),redirect:'manual',signal:AbortSignal.timeout(20000)});
  const location=response.headers.get('location');
  checks.push({path:c.path,method:c.method||'GET',origin:c.origin||null,forgedHeaders:!!c.headers,status:response.status,expectedStatus:c.status,location,expectedLocation:c.location||null,pass:response.status===c.status&&(!c.location||location===c.location)});
 }
 const out={observedAt:new Date().toISOString(),base,paidCalls:0,checks,passed:checks.filter(c=>c.pass).length,failed:checks.filter(c=>!c.pass).length};
 console.log(JSON.stringify(out));if(out.failed)process.exitCode=1;
})().catch(e=>{console.error(e.name);process.exit(1)});
